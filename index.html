AUTHENTICATION (MASUK / DAFTAR) --- window.openAuthModal = () => { document.getElementById('auth-modal').classList.remove('hidden-view'); window.switchAuthTab('login'); }; window.closeAuthModal = () => { document.getElementById('auth-modal').classList.add('hidden-view'); }; window.switchAuthTab = (tab) => { const formLogin = document.getElementById('form-login'); const formReg = document.getElementById('form-register'); const formForgot = document.getElementById('form-forgot-pwd'); const tabs = document.getElementById('auth-tabs'); const title = document.getElementById('auth-modal-title'); formLogin.classList.add('hidden-view'); formReg.classList.add('hidden-view'); formForgot.classList.add('hidden-view'); if(tab === 'login') { tabs.classList.remove('hidden-view'); document.getElementById('tab-login').className = 'flex-1 py-2 text-xs font-bold rounded-md bg-ozgold-600 text-white shadow transition-all'; document.getElementById('tab-register').className = 'flex-1 py-2 text-xs font-bold rounded-md text-slate-400 hover:text-white transition-all'; formLogin.classList.remove('hidden-view'); title.innerHTML = ' Akses Sistem'; } else if (tab === 'register') { tabs.classList.remove('hidden-view'); document.getElementById('tab-register').className = 'flex-1 py-2 text-xs font-bold rounded-md bg-ozgold-600 text-white shadow transition-all'; document.getElementById('tab-login').className = 'flex-1 py-2 text-xs font-bold rounded-md text-slate-400 hover:text-white transition-all'; formReg.classList.remove('hidden-view'); title.innerHTML = ' Daftar Warga'; } else if (tab === 'forgot') { tabs.classList.add('hidden-view'); formForgot.classList.remove('hidden-view'); title.innerHTML = ' Lupa Password'; } }; window.processRegister = async () => { const name = document.getElementById('auth-reg-name').value.trim(); const email = document.getElementById('auth-reg-email').value.trim(); const wa = document.getElementById('auth-reg-wa').value.trim(); const pwd = document.getElementById('auth-reg-pwd').value; try { if(!defDb) { alert("Sistem belum siap. Tunggu sebentar."); return; } const snapshot = await defDb.collection('users').where('email', '==', email).get(); if(!snapshot.empty) { alert("Email sudah terdaftar!"); return; } await defDb.collection('users').add({ name, email, phone: wa, password: pwd, role: 'nasabah', status: 'aktif', createdAt: firebase.firestore.FieldValue.serverTimestamp() }); window.showToast("Pendaftaran Berhasil! Silakan Masuk."); window.switchAuthTab('login'); } catch(e) { alert("Gagal mendaftar: " + e.message); } }; // PERUBAHAN: Login hanya menggunakan ID (Email/WA) window.processLogin = async () => { const id = document.getElementById('auth-login-id').value.trim(); // Bypass Super Admin (Hanya perlu memasukkan email admin) const savedEmail = localStorage.getItem('ozdigi_admin_email') || 'admin@oznet.com'; if(id === savedEmail) { window.currentUser = { id: 'admin-super', name: 'Super Admin', role: 'admin', status: 'aktif' }; window.showToast("Berhasil login sebagai Super Admin!"); window.closeAuthModal(); window.updateUIAuth(); return; } try { if(!defDb) { alert("Sistem database belum siap. Tunggu sebentar."); return; } // Cari user berdasarkan email ATAU phone, TANPA mencocokkan password let snapshot = await defDb.collection('users').where('email', '==', id).get(); if(snapshot.empty) { snapshot = await defDb.collection('users').where('phone', '==', id).get(); } if(snapshot.empty) { alert("Akun tidak terdaftar. Silakan daftar terlebih dahulu."); return; } const userData = snapshot.docs[0].data(); const userId = snapshot.docs[0].id; if(userData.status === 'blokir') { alert("Akun Anda telah dinonaktifkan oleh Admin."); return; } window.currentUser = { id: userId, ...userData }; window.showToast(`Selamat datang, ${userData.name}!`); window.closeAuthModal(); window.updateUIAuth(); } catch(e) { alert("Gagal masuk (Koneksi Error): " + e.message); } }; // FITUR BARU: Proses Lupa Password window.processForgotPwd = async () => { const email = document.getElementById('auth-forgot-email').value.trim(); if(!defDb) { alert("Sistem belum siap."); return; } try { const snapshot = await defDb.collection('users').where('email', '==', email).get(); if(snapshot.empty) { alert("Email tidak ditemukan di sistem kami."); return; } // Simulasi pemulihan password berhasil window.showToast("Akses berhasil dipulihkan! Cek email Anda."); setTimeout(() => { window.switchAuthTab('login'); }, 2000); } catch (e) { alert("Terjadi kesalahan: " + e.message); } }; window.logoutUser = () => { window.currentUser = null; window.isAdmin = false; window.showToast("Anda telah keluar."); window.updateUIAuth(); }; window.updateUIAuth = () => { window.isAdmin = window.currentUser && window.currentUser.role === 'admin'; if(window.currentUser) { // Desktop document.getElementById('nav-login-btn').classList.add('hidden-view'); // Mobile document.getElementById('mobile-login-btn').classList.add('hidden-view'); document.getElementById('mobile-logout-btn').classList.remove('hidden-view'); if(window.isAdmin) { document.getElementById('nav-admin-menu').classList.remove('hidden-view'); document.getElementById('mobile-admin-panel').classList.remove('hidden-view'); document.getElementById('btn-add-ozf').classList.remove('hidden-view'); document.getElementById('btn-add-def').classList.remove('hidden-view'); } else { document.getElementById('nav-admin-menu').classList.add('hidden-view'); document.getElementById('mobile-admin-panel').classList.add('hidden-view'); document.getElementById('btn-add-ozf').classList.add('hidden-view'); document.getElementById('btn-add-def').classList.add('hidden-view'); } } else { document.getElementById('nav-login-btn').classList.remove('hidden-view'); document.getElementById('nav-admin-menu').classList.add('hidden-view'); document.getElementById('mobile-login-btn').classList.remove('hidden-view'); document.getElementById('mobile-logout-btn').classList.add('hidden-view'); document.getElementById('mobile-admin-panel').classList.add('hidden-view'); document.getElementById('btn-add-ozf').classList.add('hidden-view'); document.getElementById('btn-add-def').classList.add('hidden-view'); } if (document.getElementById('ozf-search-input')) window.ozfSearch(); if (document.getElementById('def-search-input')) window.defSearch(); window.renderPosts(); };
